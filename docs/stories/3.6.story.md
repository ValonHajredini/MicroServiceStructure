## Story

**As a** user,
**I want** an Angular app to view and edit notes,
**so that** I have a user-friendly interface.

## Acceptance Criteria

1. Angular app at notes.mydomain.com with sidebar folder tree navigation
2. Note list view shows: title, preview (first 100 chars), last modified date, pin icon
3. Clicking note opens editor in main panel
4. Rich text editor (PrimeNG Editor or Quill.js) supports: bold, italic, lists, links
5. Auto-save triggers every 30 seconds if content changed
6. Editor includes attachment section showing uploaded files with download links
7. Responsive layout: sidebar collapses to hamburger menu on mobile

### Authentication Flow

**JWT Handling**:  
[Source: architecture/authentication-multi-tenancy.md#jwt-based-sso-flow]

```typescript
// auth.service.ts
@Injectable({ providedIn: 'root' })
export class AuthService {
  private readonly JWT_TOKEN_KEY = 'jwt_token';
  
  getToken(): string | null {
    return localStorage.getItem(this.JWT_TOKEN_KEY);
  }
  
  decodeToken(): JwtPayload | null {
    const token = this.getToken();
    if (!token) return null;
    
    try {
      const payload = JSON.parse(atob(token.split('.')[1]));
      return payload;
    } catch {
      return null;
    }
  }
  
  getTenantId(): string | null {
    return this.decodeToken()?.tenantId;
  }
  
  getUserId(): string | null {
    return this.decodeToken()?.sub;
  }
  
  isAuthenticated(): boolean {
    const payload = this.decodeToken();
    if (!payload) return false;
    
    const now = Date.now() / 1000;
    return payload.exp > now;
  }
  
  logout(): void {
    localStorage.removeItem(this.JWT_TOKEN_KEY);
    // Redirect to login
  }
}
```

**HTTP Interceptor for JWT**:

```typescript
// http-interceptor.service.ts
@Injectable()
export class JwtInterceptor implements HttpInterceptor {
  constructor(private authService: AuthService) {}
  
  intercept(req: HttpRequest<any>, next: HttpHandler): Observable<HttpEvent<any>> {
    const token = this.authService.getToken();
    
    if (token) {
      req = req.clone({
        setHeaders: {
          Authorization: `Bearer ${token}`
        }
      });
    }
    
    return next.handle(req).pipe(
      catchError((error: HttpErrorResponse) => {
        if (error.status === 401) {
          this.authService.logout();
        }
        return throwError(() => error);
      })
    );
  }
}
```

### API Service Patterns

**Notes Service**:

```typescript
// notes.service.ts
@Injectable({ providedIn: 'root' })
export class NotesService {
  private apiUrl = environment.apiUrl + '/api/v1/notes';
  
  constructor(private http: HttpClient) {}
  
  getNotes(params?: { folder_id?: string, page?: number, limit?: number }): Observable<Note[]> {
    return this.http.get<ApiResponse<Note[]>>(this.apiUrl, { params }).pipe(
      map(response => response.data)
    );
  }
  
  getNote(id: string): Observable<Note> {
    return this.http.get<ApiResponse<Note>>(`${this.apiUrl}/${id}`).pipe(
      map(response => response.data)
    );
  }
  
  createNote(note: CreateNoteDto): Observable<Note> {
    return this.http.post<ApiResponse<Note>>(this.apiUrl, note).pipe(
      map(response => response.data)
    );
  }
  
  updateNote(id: string, note: UpdateNoteDto): Observable<Note> {
    return this.http.patch<ApiResponse<Note>>(`${this.apiUrl}/${id}`, note).pipe(
      map(response => response.data)
    );
  }
  
  deleteNote(id: string): Observable<void> {
    return this.http.delete<void>(`${this.apiUrl}/${id}`);
  }
}
```

### Auto-Save Implementation

**Auto-Save with Debounce and Interval**:

```typescript
// note-editor.component.ts
export class NoteEditorComponent implements OnInit, OnDestroy {
  private autoSaveInterval = 30000; // 30 seconds
  private debounceTime = 2000; // 2 seconds
  private isDirty = false;
  private isSaving = false;
  private contentChanges$ = new Subject<void>();
  private intervalSubscription?: Subscription;
  
  ngOnInit() {
    // Debounce content changes
    this.contentChanges$.pipe(
      debounceTime(this.debounceTime)
    ).subscribe(() => {
      this.isDirty = true;
    });
    
    // Auto-save interval
    this.intervalSubscription = interval(this.autoSaveInterval).subscribe(() => {
      if (this.isDirty && !this.isSaving) {
        this.save();
      }
    });
  }
  
  onContentChange() {
    this.contentChanges$.next();
  }
  
  save() {
    if (this.isSaving) return;
    
    this.isSaving = true;
    this.saveStatus = 'Saving...';
    
    const saveOperation = this.note.id 
      ? this.notesService.updateNote(this.note.id, this.note)
      : this.notesService.createNote(this.note);
    
    saveOperation.subscribe({
      next: (saved) => {
        this.note = saved;
        this.isDirty = false;
        this.isSaving = false;
        this.saveStatus = 'All changes saved';
      },
      error: (err) => {
        this.isSaving = false;
        this.saveStatus = 'Error saving';
        this.messageService.add({
          severity: 'error',
          summary: 'Save Failed',
          detail: 'Could not save note. Will retry...'
        });
        // Retry after delay
        setTimeout(() => this.save(), 5000);
      }
    });
  }
  
  ngOnDestroy() {
    this.intervalSubscription?.unsubscribe();
    
    // Save on navigate away
    if (this.isDirty) {
      this.save();
    }
  }
}
```

### Pipes

**Relative Time Pipe**:

```typescript
@Pipe({ name: 'relativeTime' })
export class RelativeTimePipe implements PipeTransform {
  transform(value: Date): string {
    const now = new Date();
    const diff = now.getTime() - new Date(value).getTime();
    const seconds = Math.floor(diff / 1000);
    const minutes = Math.floor(seconds / 60);
    const hours = Math.floor(minutes / 60);
    const days = Math.floor(hours / 24);
    
    if (seconds < 60) return 'Just now';
    if (minutes < 60) return `${minutes} minute${minutes > 1 ? 's' : ''} ago`;
    if (hours < 24) return `${hours} hour${hours > 1 ? 's' : ''} ago`;
    if (days === 1) return 'Yesterday';
    if (days < 7) return `${days} days ago`;
    return new Date(value).toLocaleDateString();
  }
}
```

**File Size Pipe**:

```typescript
@Pipe({ name: 'fileSize' })
export class FileSizePipe implements PipeTransform {
  transform(bytes: number): string {
    if (bytes === 0) return '0 Bytes';
    const k = 1024;
    const sizes = ['Bytes', 'KB', 'MB', 'GB'];
    const i = Math.floor(Math.log(bytes) / Math.log(k));
    return Math.round(bytes / Math.pow(k, i) * 100) / 100 + ' ' + sizes[i];
  }
}
```
