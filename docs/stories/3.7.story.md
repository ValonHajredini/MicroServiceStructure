searchNotes(query: string, page: number = 1, limit: number = 20): Observable<SearchResult> {
  const params = { q: query, page: page.toString(), limit: limit.toString() };
  return this.http.get<ApiResponse<SearchResult>>(`${this.apiUrl}/search`, { params }).pipe(
    map(response => response.data)
  );

**Search Bar Component**:

```typescript
// search-bar.component.ts
export class SearchBarComponent implements OnInit {
  searchControl = new FormControl('');
  searchResults$ = new Subject<Note[]>();
  isSearching = false;
  
  ngOnInit() {
    this.searchControl.valueChanges.pipe(
      debounceTime(500),
      distinctUntilChanged(),
      switchMap(query => {
        if (!query || query.length < 1) {
          return of([]);
        }
        this.isSearching = true;
        return this.notesService.searchNotes(query).pipe(
          catchError(() => {
            this.messageService.add({
              severity: 'error',
              summary: 'Search Failed',
              detail: 'Could not search notes. Please try again.'
            });
            return of([]);
          }),
          finalize(() => this.isSearching = false)
        );
      })
    ).subscribe(results => {
      this.searchResults$.next(results);
    });
  }
  
  clearSearch() {
    this.searchControl.setValue('');
    this.searchResults$.next([]);
  }
}
```

### Filter Implementation

**Filter Types**:

1. **All Notes** (default): No filter applied
2. **Pinned Only**: Filter client-side or server-side
3. **By Folder**: Add folder_id query param to API call
4. **Recent**: Client-side filter (notes from last 7 days)

**Filter Service**:

```typescript
// filter.service.ts
@Injectable({ providedIn: 'root' })
export class FilterService {
  private activeFilter$ = new BehaviorSubject<FilterOption>({ type: 'all' });
  
  get activeFilter(): Observable<FilterOption> {
    return this.activeFilter$.asObservable();
  }
  
  setFilter(filter: FilterOption) {
    this.activeFilter$.next(filter);
  }
  
  clearFilter() {
    this.setFilter({ type: 'all' });
  }
  
  applyFilter(notes: Note[], filter: FilterOption): Note[] {
    switch (filter.type) {
      case 'pinned':
        return notes.filter(n => n.is_pinned);
      case 'folder':
        return notes; // Server-side filter via API
      case 'recent':
        const sevenDaysAgo = new Date();
        sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 7);
        return notes.filter(n => new Date(n.created_at) > sevenDaysAgo);
      case 'all':
      default:
        return notes;
    }
  }
}

interface FilterOption {
  type: 'all' | 'pinned' | 'folder' | 'recent';
  folderId?: string;
  folderName?: string;
}
```

constructor(private confirmationService: ConfirmationService) {}

deleteNote(note: Note) {
  this.confirmationService.confirm({

          this.messageService.add({
            severity: 'error',

      });

  });

### Error Handling Service

**Centralized Error Handling**:

```typescript
// error.service.ts
@Injectable({ providedIn: 'root' })
export class ErrorService {
  constructor(private messageService: MessageService) {}
  
  handleError(error: HttpErrorResponse, context?: string) {
    let message = 'An error occurred';
    
    switch (error.status) {
      case 0:
        message = 'No internet connection. Please check your connection.';
        break;
      case 400:
        message = error.error?.error?.message || 'Invalid request';
        break;
      case 401:
        message = 'Session expired. Please login again.';
        // Redirect to login
        break;
      case 403:
        message = "You don't have permission to perform this action.";
        break;
      case 404:
        message = `${context || 'Item'} not found. It may have been deleted.`;
        break;
      case 500:
      default:
        message = 'Something went wrong. Please try again.';
        break;
    }
    
    this.messageService.add({
      severity: 'error',
      summary: 'Error',
      detail: message,
      life: 5000
    });
    
    console.error('API Error:', error);
  }
}
```

### Keyboard Shortcuts

**Keyboard Service**:

```typescript
// keyboard.service.ts
@Injectable({ providedIn: 'root' })
export class KeyboardService {
  constructor() {}
  
  registerShortcuts(shortcuts: { [key: string]: () => void }) {
    fromEvent<KeyboardEvent>(document, 'keydown')
      .pipe(
        filter(event => {
          const key = `${event.ctrlKey || event.metaKey ? 'Ctrl+' : ''}${event.key}`;
          return key in shortcuts;
        })
      )
      .subscribe(event => {
        event.preventDefault();
        const key = `${event.ctrlKey || event.metaKey ? 'Ctrl+' : ''}${event.key}`;
        shortcuts[key]();
      });
  }
}
```

### State Management Considerations

**For this story**, use services with BehaviorSubjects for state management:

```typescript
// notes-state.service.ts
@Injectable({ providedIn: 'root' })
export class NotesStateService {
  private notes$ = new BehaviorSubject<Note[]>([]);
  private selectedNote$ = new BehaviorSubject<Note | undefined>(undefined);
  private loading$ = new BehaviorSubject<boolean>(false);
  
  get notes(): Observable<Note[]> {
    return this.notes$.asObservable();
  }
  
  get selectedNote(): Observable<Note | undefined> {
    return this.selectedNote$.asObservable();
  }
  
  get loading(): Observable<boolean> {
    return this.loading$.asObservable();
  }
  
  setNotes(notes: Note[]) {
    this.notes$.next(notes);
  }
  
  addNote(note: Note) {
    const current = this.notes$.value;
    this.notes$.next([note, ...current]);
  }
  
  updateNote(note: Note) {
    const current = this.notes$.value;
    const index = current.findIndex(n => n.id === note.id);
    if (index !== -1) {
      current[index] = note;
      this.notes$.next([...current]);
    }
  }
  
  deleteNote(id: string) {
    const current = this.notes$.value;
    this.notes$.next(current.filter(n => n.id !== id));
  }
  
  selectNote(note: Note | undefined) {
    this.selectedNote$.next(note);
  }
  
  setLoading(loading: boolean) {
    this.loading$.next(loading);
  }
}
```
