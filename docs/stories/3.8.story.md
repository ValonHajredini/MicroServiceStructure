# Story 3.8: Notes UI - File Upload Integration

## Status
Approved

## Story

**As a** user,
**I want** to upload files to notes via drag-and-drop or file picker,
**so that** I can attach documents easily.

## Acceptance Criteria

1. Attachment section includes "Add file" button opening file picker
2. Drag-and-drop zone accepts files dropped onto editor
3. Client calls Core Service POST /files/presigned-url to get upload URL
4. File uploaded directly to DigitalOcean Spaces using presigned URL
5. After upload, client calls POST /notes/:id/attachments with file_id
6. Upload progress indicator shows percentage complete
7. File size validation: 25MB max per file, error message if exceeded

### File Upload Flow

**Three-Step Process**:  
[Source: Story 2.6 (File Upload Service)]

2. **Upload to Spaces** - Client uploads file directly to DigitalOcean Spaces
3. **Create Attachment** - Client tells Notes Service to link file to note

### File Upload Service Implementation

**Complete Upload Flow**:

```typescript
// file-upload.service.ts
@Injectable({ providedIn: 'root' })
export class FileUploadService {
  private coreApiUrl = environment.coreApiUrl + '/api/v1/files';
  
  constructor(
    private http: HttpClient,
    private notesService: NotesService
  ) {}
  
  uploadFile(noteId: string, file: File): Observable<UploadResult> {
    return new Observable(subscriber => {
      // Step 1: Validate file size
      const MAX_SIZE = 25 * 1024 * 1024; // 25MB
      if (file.size > MAX_SIZE) {
        subscriber.error({
          message: `File size exceeds 25MB limit. File size: ${(file.size / 1024 / 1024).toFixed(2)}MB`
        });
        return;
      }
      
      // Step 2: Get presigned URL
      this.getPresignedUrl(file.name, file.type, file.size)
        .pipe(
          switchMap(urlData => {
            subscriber.next({ phase: 'uploading', progress: 0 });
            
            // Step 3: Upload to Spaces
            return this.uploadToSpaces(file, urlData.presigned_url).pipe(
              tap(progress => {
                subscriber.next({ phase: 'uploading', progress });
              }),
              switchMap(() => {
                subscriber.next({ phase: 'attaching', progress: 100 });
                
                // Step 4: Attach to note
                return this.attachFileToNote(noteId, urlData.file_id);
              })
            );
          })
        )
        .subscribe({
          next: (attachment) => {
            subscriber.next({ phase: 'complete', progress: 100, attachment });
            subscriber.complete();
          },
          error: (err) => subscriber.error(err)
        });
    });
  }
  
  private getPresignedUrl(filename: string, mimeType: string, fileSize: number): Observable<PresignedUrlResponse> {
    return this.http.post<ApiResponse<PresignedUrlResponse>>(
      `${this.coreApiUrl}/presigned-url`,
      { filename, mime_type: mimeType, file_size: fileSize }
    ).pipe(
      map(response => response.data)
    );
  }
  
  private uploadToSpaces(file: File, presignedUrl: string): Observable<number> {
    const req = new HttpRequest('PUT', presignedUrl, file, {
      headers: new HttpHeaders({
        'Content-Type': file.type
      }),
      reportProgress: true
    });
    
    return this.http.request(req).pipe(
      filter(event => event.type === HttpEventType.UploadProgress || event.type === HttpEventType.Response),
      map(event => {
        if (event.type === HttpEventType.UploadProgress) {
          return Math.round(100 * event.loaded / event.total!);
        }
        return 100;
      })
    );
  }
  
  private attachFileToNote(noteId: string, fileId: string): Observable<Attachment> {
    return this.http.post<ApiResponse<Attachment>>(
      `${environment.apiUrl}/api/v1/notes/${noteId}/attachments`,
      { file_id: fileId }
    ).pipe(
      map(response => response.data)
    );
  }
}

interface PresignedUrlResponse {
  presigned_url: string;
  file_id: string;
  storage_key: string;
}

interface UploadResult {
  phase: 'uploading' | 'attaching' | 'complete';
  progress: number;
  attachment?: Attachment;
}
```

### File Upload Component

**Component Implementation**:

```typescript
// file-upload.component.ts
@Component({
  selector: 'app-file-upload',
  templateUrl: './file-upload.component.html',
  styleUrls: ['./file-upload.component.scss']
})
export class FileUploadComponent {
  @Input() noteId!: string;
  @Output() fileUploaded = new EventEmitter<Attachment>();
  
  @ViewChild('fileInput') fileInput!: ElementRef<HTMLInputElement>;
  
  isUploading = false;
  uploadProgress = 0;
  uploadPhase: string = '';
  isDragOver = false;
  
  constructor(
    private fileUploadService: FileUploadService,
    private messageService: MessageService
  ) {}
  
  onFileSelected(event: Event) {
    const input = event.target as HTMLInputElement;
    if (input.files && input.files.length > 0) {
      this.uploadFiles(Array.from(input.files));
    }
  }
  
  onDragOver(event: DragEvent) {
    event.preventDefault();
    this.isDragOver = true;
  }
  
  onDragLeave(event: DragEvent) {
    event.preventDefault();
    this.isDragOver = false;
  }
  
  onDrop(event: DragEvent) {
    event.preventDefault();
    this.isDragOver = false;
    
    if (event.dataTransfer?.files) {
      this.uploadFiles(Array.from(event.dataTransfer.files));
    }
  }
  
  openFilePicker() {
    this.fileInput.nativeElement.click();
  }
  
  uploadFiles(files: File[]) {
    // Validate file sizes
    const MAX_SIZE = 25 * 1024 * 1024;
    const invalidFiles = files.filter(f => f.size > MAX_SIZE);
    
    if (invalidFiles.length > 0) {
      this.messageService.add({
        severity: 'error',
        summary: 'File Too Large',
        detail: `${invalidFiles.length} file(s) exceed 25MB limit`
      });
    }
    
    const validFiles = files.filter(f => f.size <= MAX_SIZE);
    
    // Upload each file
    validFiles.forEach(file => this.uploadFile(file));
  }
  
  uploadFile(file: File) {
    this.isUploading = true;
    this.uploadProgress = 0;
    this.uploadPhase = 'Preparing...';
    
    this.fileUploadService.uploadFile(this.noteId, file).subscribe({
      next: (result) => {
        if (result.phase === 'uploading') {
          this.uploadPhase = 'Uploading...';
          this.uploadProgress = result.progress;
        } else if (result.phase === 'attaching') {
          this.uploadPhase = 'Attaching to note...';
        } else if (result.phase === 'complete' && result.attachment) {
          this.isUploading = false;
          this.uploadProgress = 100;
          this.uploadPhase = '';
          this.fileUploaded.emit(result.attachment);
          this.messageService.add({
            severity: 'success',
            summary: 'Success',
            detail: 'File attached successfully'
          });
        }
      },
      error: (err) => {
        this.isUploading = false;
        this.uploadProgress = 0;
        this.uploadPhase = '';
        this.messageService.add({
          severity: 'error',
          summary: 'Upload Failed',
          detail: err.message || 'Could not upload file'
        });
      }
    });
  }
}
```

### File Size Validation

**Client-Side Validation**:

```typescript
const MAX_FILE_SIZE = 25 * 1024 * 1024; // 25MB
const MAX_TOTAL_SIZE = 100 * 1024 * 1024; // 100MB

function validateFileSize(file: File): boolean {
  return file.size <= MAX_FILE_SIZE;
}

function validateTotalSize(currentTotal: number, newFileSize: number): boolean {
  return (currentTotal + newFileSize) <= MAX_TOTAL_SIZE;
}
```

**Error Messages**:

```typescript
// File too large
```
